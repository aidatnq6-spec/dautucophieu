**Goal:** From the 15:30 slot of each trading session, build a board of MACD (12-26-9) state and RSI (14) values over the frames M, W, D1, H4 and H1 for 17 instruments. Show it on a new page **Xu hướng**, highlighting the cells that changed since the previous session's board.

**Architecture:**
- Closes are fetched into small JSON series stores under `data/trend-sources` (daily) and `data/intraday` (hourly):
  - Vnstock for Vietnamese stocks and indices;
  - Coinbase public candles for BTC and PAXG;
  - FRED CSV for the dollar index, the US 10-year yield and WTI.
- Pure functions resample those closes into W/M/H4 and compute indicators the way TradingView does by default.
- A snapshot per session is saved under `data/trend/<date>.json`, already compared with the previous snapshot.
- FastAPI serves the latest snapshot. A Next page renders it.

**Tech Stack:** Python 3.11 (FastAPI, pydantic, pandas only inside the Vnstock adapter), Next.js 16 / React client components, Playwright e2e, pytest, PowerShell 5.1 scheduler script.

```python
"""MACD and RSI as TradingView computes them by default, and bar resampling for the trend board.

EMA uses alpha 2/(n+1) seeded with the simple average of the first n values (Pine ta.ema); RSI
smooths gains and losses with Wilder's average seeded the same way (Pine ta.rsi). Indicators are
ratios of prices, not money, so floats are used here. Points are (datetime, close) sorted ascending.
"""

from datetime import timezone
from decimal import ROUND_HALF_UP, Decimal

MIN_MACD_BARS, MIN_RSI_BARS = 35, 15


def ema(values: list[float], n: int) -> list[float | None]:
    out: list[float | None] = [None] * len(values)
    if len(values) < n:
        return out
    prev = sum(values[:n]) / n
    out[n - 1] = prev
    alpha = 2 / (n + 1)
    for i in range(n, len(values)):
        prev = alpha * values[i] + (1 - alpha) * prev
        out[i] = prev
    return out


def macd(closes: list[float], fast: int = 12, slow: int = 26, signal: int = 9) -> tuple[list[float | None], list[float | None]]:
    line = [a - b if a is not None and b is not None else None for a, b in zip(ema(closes, fast), ema(closes, slow))]
    start = next((i for i, v in enumerate(line) if v is not None), len(line))
    return line, [None] * start + ema(line[start:], signal)


def rsi(closes: list[float], n: int = 14) -> list[float | None]:
    out: list[float | None] = [None] * len(closes)
    if len(closes) < n + 1:
        return out
    gains = [max(closes[i] - closes[i - 1], 0.0) for i in range(1, len(closes))]
    losses = [max(closes[i - 1] - closes[i], 0.0) for i in range(1, len(closes))]

    def value(g, l):
        return 100.0 if l == 0 else 0.0 if g == 0 else 100 - 100 / (1 + g / l)
    avg_g, avg_l = sum(gains[:n]) / n, sum(losses[:n]) / n
    out[n] = value(avg_g, avg_l)
    for i in range(n + 1, len(closes)):
        avg_g = (avg_g * (n - 1) + gains[i - 1]) / n
        avg_l = (avg_l * (n - 1) + losses[i - 1]) / n
        out[i] = value(avg_g, avg_l)
    return out


def macd_state(closes: list[float]) -> str | None:
    """"up"/"down": MACD above/below its signal on the last bar; ">0"/"<0": the MACD line's sign."""
    if len(closes) < MIN_MACD_BARS:
        return None
    line, signal = macd(closes)
    return f"{'up' if line[-1] > signal[-1] else 'down'},{'>0' if line[-1] > 0 else '<0'}"


def rsi_value(closes: list[float]) -> int | None:
    if len(closes) < MIN_RSI_BARS:
        return None
    return int(Decimal(str(rsi(closes)[-1])).quantize(Decimal(1), ROUND_HALF_UP))


def _last_per(points, key) -> list[float]:
    out, last = [], object()
    for t, v in points:
        k = key(t)
        if out and k == last:
            out[-1] = v
        else:
            out.append(v)
            last = k
    return out


def weekly(points) -> list[float]:
    return _last_per(points, lambda t: t.isocalendar()[:2])


def monthly(points) -> list[float]:
    return _last_per(points, lambda t: (t.year, t.month))


def four_hourly(points) -> list[float]:
    """Four-hour bars anchored at 00:00 UTC (07:00 in Vietnam), as TradingView draws them."""
    def key(t):
        u = t.astimezone(timezone.utc)
        return u.date(), u.hour // 4
    return _last_per(points, key)
```

```python
"""Close series keyed by timestamp, one JSON file per key (e.g. "FPT-1h", "BTC-1d").

Values are kept as decimal strings and read back as floats for indicator math. A newer value at
the same timestamp replaces the older one; only the latest `keep` points are retained.
"""

import json
import os
import re
from datetime import datetime
from pathlib import Path

_KEY = re.compile(r"^[A-Za-z0-9_-]{1,60}$")


class SeriesStore:
    def __init__(self, root: Path):
        self.root = Path(root)

    def _path(self, key: str) -> Path:
        if not _KEY.fullmatch(key):
            raise ValueError("invalid series key")
        return self.root / f"{key}.json"

    def read(self, key: str) -> list[tuple[datetime, float]]:
        path = self._path(key)
        if not path.is_file():
            return []
        return [(datetime.fromisoformat(t), float(v)) for t, v in json.loads(path.read_text(encoding="utf-8"))["points"]]

    def upsert(self, key: str, points: list[tuple[datetime, str]], *, keep: int = 6000) -> None:
        path = self._path(key)
        merged = {t: v for t, v in (json.loads(path.read_text(encoding="utf-8"))["points"] if path.is_file() else [])}
        for t, v in points:
            stamp = t.date().isoformat() if key.endswith("-1d") else t.isoformat()
            merged[stamp] = str(v)
        ordered = sorted(merged.items(), key=lambda kv: datetime.fromisoformat(kv[0]))[-keep:]
        self.root.mkdir(parents=True, exist_ok=True)
        tmp = path.with_suffix(".tmp")
        tmp.write_text(json.dumps({"key": key, "points": ordered}), encoding="utf-8")
        os.replace(tmp, path)
```

Note: daily keys end in `-1d` and are stored as dates. Ordering mixes naive dates and aware hours only within a single key, never across keys.

    def fetch_closes(self, symbol: str, start: date, end: date, *, interval: str, index: bool) -> list[tuple[datetime, str]]:
        """Closes only, for indicators ("1D" or "1H"). Units follow the source (thousand đồng for stocks, points for
        indices); MACD direction/sign and RSI do not depend on the unit."""
        market, _ = self._api()
        handle = market.index(symbol.upper()) if index else market.equity(symbol.upper())
        frame = handle.ohlcv(start=start.isoformat(), end=end.isoformat(), interval=interval)
        out = []
        for row in frame.to_dict("records"):
            try:
                value = Decimal(str(row["close"]))
            except (InvalidOperation, KeyError):
                continue
            if value.is_finite() and value > 0:
                out.append((row["time"].to_pydatetime(), format(value.normalize(), "f")))
        return out
```

"""Public close feeds for the trend board's non-Vietnamese rows.

Coinbase Exchange public candles (no account): at most 300 candles per call, newest first, rows
[time, low, high, open, close, volume]. FRED public CSV: one close per day.
"""

import json
from datetime import date, datetime, timedelta, timezone
from decimal import Decimal

from .leading import FRED_CSV, parse_fred_csv

COINBASE = "https://api.exchange.coinbase.com/products/{product}/candles?granularity={g}&start={s}&end={e}"


def coinbase_closes(get, product: str, granularity: int, start: date, end: date) -> list[tuple[datetime, str]]:
    span = timedelta(seconds=granularity * 300)
    t0 = datetime.combine(start, datetime.min.time(), tzinfo=timezone.utc)
    t_end = datetime.combine(end, datetime.min.time(), tzinfo=timezone.utc) + timedelta(days=1)
    seen: dict[datetime, str] = {}
    while t0 < t_end:
        t1 = min(t0 + span, t_end)
        url = COINBASE.format(product=product, g=granularity, s=t0.isoformat().replace("+00:00", "Z"), e=t1.isoformat().replace("+00:00", "Z"))
        for row in json.loads(get(url)):
            seen[datetime.fromtimestamp(int(row[0]), timezone.utc)] = format(Decimal(str(row[4])).normalize(), "f")
        t0 = t1
    return sorted(seen.items())


def fred_closes(get, series: str, start: date) -> list[tuple[datetime, str]]:
    text = get(FRED_CSV.format(series=series) + f"&cosd={start.isoformat()}")
    return [(datetime(d.year, d.month, d.day), v) for d, v in parse_fred_csv(text)]
```

    return [(datetime(d.year, d.month, d.day), 100.0 + (i if up else -i)) for i, d in enumerate(days)]


def hourly(n, end=date(2026, 10, 23)):
    out, day = [], end
    while len(out) < n:
        if CAL.is_session(day):
            out = [(datetime(day.year, day.month, day.day, h, tzinfo=VN), 0.0) for h in (9, 10, 11, 13, 14)] + out
        day -= timedelta(days=1)
    return [(t, 100.0 + i) for i, (t, _) in enumerate(out[-n:])]

```python
"""Trend board: MACD state and RSI over M, W, D1, H4, H1 for the owner's 17 instruments.

Closes come from Vnstock (Vietnamese stocks and indices), Coinbase public candles (BTC, PAXG as a
gold proxy) and FRED (dollar index, US 10-year, WTI; daily only). One snapshot per session, built
from the 15:30 slot on, is compared with the previous saved snapshot so changed cells can be
highlighted. Missing cells carry a reason; nothing is interpolated. Not a recommendation.
"""

import json
import os
from datetime import date, datetime, time, timedelta, timezone
from pathlib import Path

from investment_agent.domain.pinned import CONFIG_DIR

from .indicators import four_hourly, macd_state, monthly, rsi_value, weekly
from .series import SeriesStore
from .trend_sources import coinbase_closes, fred_closes

BOARD_PATH = CONFIG_DIR / "trend-board.json"
FRAMES = ("M", "W", "D1", "H4", "H1")
VN = timezone(timedelta(hours=7))
CLOSE = time(15, 30)
FULL_DAYS, REFRESH_DAYS, HOURLY_FULL_DAYS, HOURLY_REFRESH_DAYS = 2920, 60, 365, 30


def load_board(path: Path = BOARD_PATH) -> list[dict]:
    return json.loads(Path(path).read_text(encoding="utf-8"))["groups"]


def _version(path: Path = BOARD_PATH) -> str:
    return json.loads(Path(path).read_text(encoding="utf-8"))["version"]


def zone(value: int | None) -> str | None:
    return None if value is None else "overbought" if value >= 70 else "oversold" if value <= 30 else "neutral"


def _cell(closes: list[float] | None, reason: str | None = None) -> dict:
    if reason is None and not closes:
        reason = "Chưa có dữ liệu"
    if reason is not None:
        return {"macd": None, "rsi": None, "zone": None, "reason": reason}
    state, value = macd_state(closes), rsi_value(closes)
    return {"macd": state, "rsi": value, "zone": zone(value), "reason": None if state or value is not None else "Chưa đủ lịch sử"}


def build_cells(daily, hourly, *, session_date: date | None, hourly_supported: bool) -> dict:
    """daily/hourly: (datetime, close) ascending. session_date set for Vietnamese rows: their last bar must be that day."""
    def fresh(points):
        return session_date is None or (bool(points) and points[-1][0].date() == session_date)
    d_reason = None if not daily or fresh(daily) else "Chưa có giá của phiên"
    cells = {"D1": _cell([v for _, v in daily or []], d_reason), "W": _cell(weekly(daily or []), d_reason),
             "M": _cell(monthly(daily or []), d_reason)}
    if not hourly_supported:  # separate dicts: compare() mutates each cell
        cells["H4"], cells["H1"] = _cell(None, "Nguồn chỉ có giá ngày"), _cell(None, "Nguồn chỉ có giá ngày")
    else:
        h_reason = None if not hourly or fresh(hourly) else "Chưa có giá của phiên"
        cells["H4"] = _cell(four_hourly(hourly or []), h_reason)
        cells["H1"] = _cell([v for _, v in hourly or []], h_reason)
    return {f: cells[f] for f in FRAMES}


def compare(snapshot: dict, previous: dict | None) -> dict:
    prev_cells = {r["id"]: r["cells"] for g in (previous or {}).get("groups", []) for r in g["rows"]}
    for g in snapshot["groups"]:
        for r in g["rows"]:
            for frame, c in r["cells"].items():
                p = prev_cells.get(r["id"], {}).get(frame) or {}
                macd_changed = bool(c["macd"] and p.get("macd") and c["macd"] != p["macd"])
                rsi_changed = bool(c["zone"] and p.get("zone") and c["zone"] != p["zone"])
                c.update({"macd_changed": macd_changed, "previous_macd": p.get("macd") if macd_changed else None,
                          "rsi_changed": rsi_changed, "previous_zone": p.get("zone") if rsi_changed else None})
    snapshot["previous_date"] = previous.get("session_date") if previous else None
    return snapshot


class SnapshotStore:
    def __init__(self, root: Path):
        self.root = Path(root)

    def _files(self):
        return sorted(self.root.glob("????-??-??.json")) if self.root.is_dir() else []

    def save(self, snapshot: dict) -> None:
        self.root.mkdir(parents=True, exist_ok=True)
        path = self.root / f"{date.fromisoformat(snapshot['session_date']).isoformat()}.json"
        tmp = path.with_suffix(".tmp")
        tmp.write_text(json.dumps(snapshot, ensure_ascii=False), encoding="utf-8")
        os.replace(tmp, path)

    def _load(self, path):
        return json.loads(path.read_text(encoding="utf-8"))

    def latest(self) -> dict | None:
        files = self._files()
        return self._load(files[-1]) if files else None

    def before(self, day: date) -> dict | None:
        """The most recent snapshot of an earlier session (normally the previous trading session)."""
        earlier = [p for p in self._files() if p.stem < day.isoformat()]
        return self._load(earlier[-1]) if earlier else None

    def prune(self, today: date, *, keep_days: int = 90) -> None:
        cutoff = (today - timedelta(days=keep_days)).isoformat()
        for p in self._files():
            if p.stem < cutoff:
                p.unlink()


def _refresh(store: SeriesStore, key: str, fetch, *, today: date, full: int, refresh: int, enough: int):
    start = today - timedelta(days=full if len(store.read(key)) < enough else refresh)
    store.upsert(key, fetch(start))


def run_trend(data_dir: Path, cal, *, now: datetime, provider, get) -> dict:
    local = now.astimezone(VN)
    today = local.date()
    try:
        if not cal.is_session(today):
            return {"status": "skipped", "reason": "not_a_session"}
    except ValueError:
        return {"status": "skipped", "reason": "calendar_missing"}
    if local.time() < CLOSE:
        return {"status": "skipped", "reason": "before_close"}
    data_dir = Path(data_dir)
    days, hours = SeriesStore(data_dir / "trend-sources"), SeriesStore(data_dir / "intraday")
    errors: dict[str, str] = {}
    groups = []
    for group in load_board():
        rows = []
        for row in group["rows"]:
            src, rid = row["source"], row["id"]
            try:
                if src["kind"] == "vn":
                    _refresh(days, f"{rid}-1d", lambda s: provider.fetch_closes(rid, s, today, interval="1D", index=src["index"]),
                             today=today, full=FULL_DAYS, refresh=REFRESH_DAYS, enough=400)
                    _refresh(hours, f"{rid}-1h", lambda s: provider.fetch_closes(rid, s, today, interval="1H", index=src["index"]),
                             today=today, full=HOURLY_FULL_DAYS, refresh=HOURLY_REFRESH_DAYS, enough=1)
                    cells = build_cells(days.read(f"{rid}-1d"), hours.read(f"{rid}-1h"), session_date=today, hourly_supported=True)
                elif src["kind"] == "coinbase":
                    _refresh(days, f"{rid}-1d", lambda s: coinbase_closes(get, src["product"], 86400, s, today),
                             today=today, full=FULL_DAYS, refresh=REFRESH_DAYS, enough=400)
                    _refresh(hours, f"{rid}-1h", lambda s: coinbase_closes(get, src["product"], 3600, s, today),
                             today=today, full=REFRESH_DAYS, refresh=7, enough=1)
                    cells = build_cells(days.read(f"{rid}-1d"), hours.read(f"{rid}-1h"), session_date=None, hourly_supported=True)
                elif src["kind"] == "fred":
                    days.upsert(f"{rid}-1d", fred_closes(get, src["series"], today - timedelta(days=FULL_DAYS)))
                    cells = build_cells(days.read(f"{rid}-1d"), None, session_date=None, hourly_supported=False)
                else:
                    cells = {f: _cell(None, src["reason"]) for f in FRAMES}
            except Exception as exc:  # one failing source never stops the board
                errors[rid] = f"error:{type(exc).__name__}"
                cells = {f: _cell(None, "Lỗi nguồn") for f in FRAMES}
            rows.append({"id": rid, "label": row["label"], "note": row.get("note"), "cells": cells})
        groups.append({"id": group["id"], "label": group["label"], "rows": rows})
    store = SnapshotStore(data_dir / "trend")
    snapshot = compare({"session_date": today.isoformat(), "generated_at": now.astimezone(timezone.utc).isoformat(),
                        "board_version": _version(), "errors": errors, "groups": groups}, store.before(today))
    store.save(snapshot)
    store.prune(today)
    return {"status": "ok", "session_date": today.isoformat(), "errors": errors}
```

"use client";
import { useEffect, useState } from "react";
import { ApiError, api } from "../lib/api";

type Cell = { macd: string | null; rsi: number | null; zone: string | null; reason: string | null;
  macd_changed?: boolean; previous_macd?: string | null; rsi_changed?: boolean; previous_zone?: string | null };
type Row = { id: string; label: string; note: string | null; cells: Record<string, Cell> };
type Snapshot = { session_date: string; generated_at: string; previous_date: string | null; errors: Record<string, string>;
  groups: { id: string; label: string; rows: Row[] }[] };

const FRAMES = ["M", "W", "D1", "H4", "H1"];
const ZONE: Record<string, string> = { overbought: "quá mua", oversold: "quá bán", neutral: "trung tính" };

function MacdCell({ c }: { c: Cell }) {
  if (!c.macd) return <td className="trend-gap">—{c.reason && <div className="caption">{c.reason}</div>}</td>;
  return (
    <td className={`trend-macd ${c.macd.startsWith("up") ? "up" : "down"}${c.macd_changed ? " changed" : ""}`}>
      {c.macd}{c.macd_changed && c.previous_macd && <div className="caption">trước: {c.previous_macd}</div>}
    </td>
  );
}

function RsiCell({ c }: { c: Cell }) {
  if (c.rsi === null) return <td className="trend-gap">—</td>;
  return (
    <td className={`num trend-rsi ${c.zone ?? ""}${c.rsi_changed ? " changed" : ""}`}>
      {c.rsi}{c.rsi_changed && c.previous_zone && <div className="caption">trước: {ZONE[c.previous_zone] ?? c.previous_zone}</div>}
    </td>
  );
}

function Card({ row }: { row: Row }) {
  return (
    <div className="trend-card">
      <h3>{row.label}</h3>
      <div className="tablewrap">
        <table aria-label={row.label}>
          <thead><tr><th>Chỉ báo</th>{FRAMES.map((f) => <th key={f}>{f}</th>)}</tr></thead>
          <tbody>
            <tr><th scope="row">MACD</th>{FRAMES.map((f) => <MacdCell key={f} c={row.cells[f]} />)}</tr>
            <tr><th scope="row">RSI</th>{FRAMES.map((f) => <RsiCell key={f} c={row.cells[f]} />)}</tr>
          </tbody>
        </table>
      </div>
      {row.note && <p className="caption">{row.note}</p>}
    </div>
  );
}

export function TrendView() {
  const [data, setData] = useState<Snapshot | null | undefined>(undefined);
  const [error, setError] = useState<string | null>(null);
  useEffect(() => { api<{ snapshot: Snapshot | null }>("trend").then((d) => setData(d.snapshot), (e) => setError(e instanceof ApiError ? e.failure.message : "Lỗi tải.")); }, []);
  if (error) return <p className="errors" role="alert">{error}</p>;
  if (data === undefined) return <p className="muted">Đang tải…</p>;
  if (data === null) return <p className="muted">Chưa có bảng: bảng được lập tự động từ 15:30 mỗi phiên giao dịch.</p>;
  return (
    <>
      <p className="notice">Phiên {data.session_date} · lập lúc {new Date(data.generated_at).toLocaleString("vi-VN")} ·{" "}
        {data.previous_date ? `ô viền đậm là ô đổi trạng thái so với phiên ${data.previous_date}` : "chưa có phiên trước để so sánh"}.
        MACD 12-26-9 (up/down: so với đường tín hiệu; &gt;0/&lt;0: so với mức 0), RSI 14. Chỉ để theo dõi, không phải khuyến nghị.</p>
      {data.groups.map((g) => (
        <section key={g.id} className="surface" aria-label={g.label}>
          <h2>{g.label}</h2>
          <div className="trend-grid">{g.rows.map((r) => <Card key={r.id} row={r} />)}</div>
        </section>
      ))}
    </>
  );
}
